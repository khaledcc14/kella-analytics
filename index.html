function loadMatchesForDate(dateStr) { const container = document.getElementById('matchesList'); if (!container) return; container.innerHTML = `
🔄 جاري جلب المباريات ليوم ${dateStr}...
`; // استبدل الرابط أدناه برابط السيرفر الخاص بك لاحقاً عند الاستضافة (أو دعه محلياً للتجربة) const serverUrl = `http://localhost:3000/api/matches/${dateStr}`; fetch(serverUrl) .then(response => response.json()) .then(data => { let matches = data.matches || []; let html = `
✅ جدول مباريات تاريخ: ${dateStr} (${matches.length} مباراة)
`; if (matches.length === 0) { html += `
لا توجد مباريات مسجلة لهذا اليوم.
`; } else { matches.forEach((m) => { html += `
🏆 ${m.league}
${m.home} vs ${m.away} ⏰ ${m.time}
🎯 تحليل الـ AI: ${m.score} | 💡 ${m.ai}
📊 أسواق الرهان اللحظية (Odds): ${m.odds}
`; }); } container.innerHTML = html; }) .catch(error => { container.innerHTML = `
⚠️ تنبيه: تأكد من تشغيل السيرفر المحلي أو السحابي لربط البيانات.
`; }); }
