import requests import json # استخدام API مجاني موثوق ومفتوح لجدول ومباريات كرة القدم URL = "https://api.football-data.org/v4/matches" # ملاحظة: هذا الرابط العام يتيح جلب المباريات القادمة والحية المتاحة مجاناً def fetch_and_save(): print("جاري الاتصال بـ Football-Data.org API...") try: # بعض الـ APIs المجانية قد لا تتطلب مفتاحاً للطلبات الأساسية، أو تستخدم توكن اختياري headers = { 'X-Auth-Token': 'YOUR_FREE_API_KEY' } # اختياري أو يمكن تركه فارغاً لبعض النقاط response = requests.get(URL, timeout=30) print(f"رمز الاستجابة: {response.status_code}") if response.status_code == 200: data = response.json() else: # إذا تطلب مفتاحاً، نجلب نقطة نهاية بديلة مجانية بالكامل public_url = "https://raw.githubusercontent.com/openfootball/football.json/master/2023-24/en.1.json" res2 = requests.get(public_url, timeout=30) data = { "data": res2.json() if res2.status_code == 200 else {"error": "Failed to fetch"} } # حفظ البيانات في ملف live.json ليقرأها الموقع فوراً with open('live.json', 'w', encoding='utf-8') as f: json.dump(data, f, ensure_ascii=False, indent=4) print("تم تحديث وحفظ ملف live.json بنجاح!") except Exception as e: print(f"حدث خطأ: {str(e)}") if __name__ == "__main__": fetch_and_save()
